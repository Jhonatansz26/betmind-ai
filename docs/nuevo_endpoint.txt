# apps/api/schemas/ticket.py
from pydantic import BaseModel, Field
from enum import Enum


class TicketMode(str, Enum):
    EDGE = "edge"
    VALUE = "value"
    BOLD = "bold"


class TicketLegSchema(BaseModel):
    match_id: int
    home_team: str
    away_team: str
    league: str
    market_name: str          # "OVER_2_5", "1X2_HOME", "BTTS_YES", etc.
    market_label: str         # "Over 2.5 Goals", "Home Win", "BTTS Yes"
    our_probability: float
    bookmaker_odds: float
    implied_probability: float
    edge_percentage: float    # our_prob - implied_prob, en porcentaje
    expected_value: float
    match_time_cot: str       # "3:00 PM COT"


class GeneratedTicket(BaseModel):
    mode: TicketMode
    mode_label: str           # "EDGE MODE", "VALUE MODE", "BOLD MODE"
    legs: list[TicketLegSchema]
    combined_odds: float
    average_ev: float
    confidence_score: int
    correlation_validated: bool
    tactical_summary: str     # Una frase del análisis de Groq
    pros: list[str]           # 2-3 puntos a favor del ticket global
    cons: list[str]           # 1-2 riesgos del ticket global
    staking_suggestion: str   # "1-2% of bankroll" según el modo


class TicketGenerateRequest(BaseModel):
    modes: list[TicketMode] = Field(
        default=[TicketMode.EDGE, TicketMode.VALUE, TicketMode.BOLD]
    )
    league_filter: list[str] | None = None  # None = todas las ligas
    date: str | None = None                 # None = hoy en COT


class TicketGenerateResponse(BaseModel):
    generated_at: str         # ISO timestamp COT
    tickets: list[GeneratedTicket]
    total_ev_opportunities: int
    matches_analyzed: int
```

# apps/api/engine/ticket_builder.py
"""
SRP: Lógica pura de construcción de tickets.
Recibe lista de predicciones ya calculadas, aplica reglas y retorna tickets.
Sin I/O — testeable de forma aislada.
"""
from dataclasses import dataclass
from apps.api.schemas.ticket import TicketMode, TicketLegSchema, GeneratedTicket

# ── Configuración por modo ─────────────────────────────────────────────────────

MODE_CONFIG = {
    TicketMode.EDGE: {
        "min_ev":              0.05,
        "max_selections":      3,
        "min_our_probability": 0.55,
        "target_odds_min":     1.40,
        "target_odds_max":     2.30,
        "allowed_markets": {
            "1X2_HOME", "1X2_AWAY",
            "OVER_2_5", "OVER_1_5",
            "BTTS_YES",
        },
        "staking": "1-2% of bankroll — conservative, high-frequency play",
    },
    TicketMode.VALUE: {
        "min_ev":              0.08,
        "max_selections":      4,
        "min_our_probability": 0.46,
        "target_odds_min":     1.90,
        "target_odds_max":     4.50,
        "allowed_markets": {
            "1X2_HOME", "1X2_AWAY", "1X2_DRAW",
            "OVER_2_5", "OVER_1_5", "OVER_3_5",
            "BTTS_YES",
            "CORNERS_OVER", "CARDS_OVER",
        },
        "staking": "0.5-1% of bankroll — medium frequency, higher EV target",
    },
    TicketMode.BOLD: {
        "min_ev":              0.03,
        "max_selections":      4,
        "min_our_probability": 0.40,
        "target_odds_min":     4.00,
        "target_odds_max":     14.00,
        "allowed_markets":     None,   # None = todos los mercados
        "require_correlation": True,   # Más estricto en este modo
        "staking": "0.25-0.5% of bankroll — low frequency, high variance",
    },
}

# ── Combinaciones prohibidas (correlación negativa) ───────────────────────────
# Formato: frozenset de market_names — orden no importa

FORBIDDEN_COMBINATIONS: list[frozenset] = [
    frozenset({"UNDER_2_5",  "BTTS_YES"}),     # Pocos goles + ambos anotan: contradictorio
    frozenset({"UNDER_1_5",  "BTTS_YES"}),     # Menos de 2 goles + ambos anotan: imposible casi
    frozenset({"OVER_3_5",   "CARDS_UNDER"}),  # Partido abierto → más tarjetas, no menos
    frozenset({"1X2_DRAW",   "BTTS_NO"}),      # Empate sin goles: muy raro, no base de ticket
    frozenset({"OVER_2_5",   "CARDS_UNDER"}),  # Alta goles → alta tensión → más tarjetas
    frozenset({"1X2_AWAY",   "CORNERS_OVER"}), # Visitante gana controlando → menos córneres del local
]

# ── Combinaciones con correlación POSITIVA (bonus de confianza) ───────────────

POSITIVE_CORRELATIONS: list[tuple[frozenset, float]] = [
    (frozenset({"1X2_HOME",  "OVER_1_5"}),    0.72),  # Local gana → al menos 2 goles
    (frozenset({"1X2_HOME",  "CORNERS_OVER"}), 0.65),  # Local dominante → más córneres
    (frozenset({"BTTS_YES",  "OVER_2_5"}),    0.81),  # Ambos anotan → suele haber +2.5
    (frozenset({"CARDS_OVER","1X2_DRAW"}),    0.58),  # Derbis igualados → más tarjetas
    (frozenset({"OVER_3_5",  "BTTS_YES"}),    0.76),  # Muchos goles → casi seguro ambos anotan
]


def check_forbidden_combination(
    selected_markets: list[str],
) -> tuple[bool, str | None]:
    """
    Verifica si la selección actual contiene una combinación prohibida.
    Retorna (is_valid, reason_if_invalid).
    """
    market_set = set(selected_markets)
    for forbidden in FORBIDDEN_COMBINATIONS:
        if forbidden.issubset(market_set):
            markets_str = " + ".join(forbidden)
            return False, f"Negative correlation detected: {markets_str}"
    return True, None


def get_correlation_bonus(selected_markets: list[str]) -> float:
    """
    Retorna un bonus de confianza (0.0 a 1.0) si hay correlaciones positivas.
    0.0 = sin correlación detectada, 1.0 = correlación máxima.
    """
    market_set = set(selected_markets)
    max_correlation = 0.0
    for corr_set, corr_value in POSITIVE_CORRELATIONS:
        if corr_set.issubset(market_set):
            max_correlation = max(max_correlation, corr_value)
    return max_correlation


def calculate_combined_odds(legs: list[TicketLegSchema]) -> float:
    """Cuota combinada = producto de todas las cuotas individuales."""
    result = 1.0
    for leg in legs:
        result *= leg.bookmaker_odds
    return round(result, 2)


def calculate_average_ev(legs: list[TicketLegSchema]) -> float:
    """EV promedio del ticket (no el EV real del parlay — eso sería negativo casi siempre)."""
    if not legs:
        return 0.0
    return round(sum(leg.expected_value for leg in legs) / len(legs), 4)


def build_ticket_for_mode(
    mode: TicketMode,
    available_predictions: list[dict],
) -> GeneratedTicket | None:
    """
    Construye el mejor ticket posible para un modo dado.

    Args:
        available_predictions: Lista de dicts con todos los mercados del día
                               que ya tienen EV calculado. Formato:
                               {
                                 "match_id": int,
                                 "home_team": str,
                                 "away_team": str,
                                 "league": str,
                                 "match_time_cot": str,
                                 "markets": [
                                   {
                                     "market_name": str,
                                     "market_label": str,
                                     "our_probability": float,
                                     "bookmaker_odds": float,
                                     "implied_probability": float,
                                     "expected_value": float,
                                   }
                                 ]
                               }
    """
    config = MODE_CONFIG[mode]
    allowed = config.get("allowed_markets")
    min_ev = config["min_ev"]
    min_prob = config["min_our_probability"]
    max_legs = config["max_selections"]
    target_min = config["target_odds_min"]
    target_max = config["target_odds_max"]

    # 1. Recolectar todos los mercados candidatos que pasan los filtros básicos
    candidates: list[TicketLegSchema] = []

    for pred in available_predictions:
        for mkt in pred.get("markets", []):
            mkt_name = mkt["market_name"]

            # Filtro por mercados permitidos en este modo
            if allowed and mkt_name not in allowed:
                continue

            # Filtro por EV mínimo
            if mkt.get("expected_value", 0) < min_ev:
                continue

            # Filtro por probabilidad mínima
            if mkt.get("our_probability", 0) < min_prob:
                continue

            edge_pct = round(
                (mkt["our_probability"] - mkt["implied_probability"]) * 100, 2
            )

            candidates.append(TicketLegSchema(
                match_id=pred["match_id"],
                home_team=pred["home_team"],
                away_team=pred["away_team"],
                league=pred["league"],
                market_name=mkt_name,
                market_label=mkt["market_label"],
                our_probability=mkt["our_probability"],
                bookmaker_odds=mkt["bookmaker_odds"],
                implied_probability=mkt["implied_probability"],
                edge_percentage=edge_pct,
                expected_value=mkt["expected_value"],
                match_time_cot=pred["match_time_cot"],
            ))

    if len(candidates) < 2:
        return None  # No hay suficientes mercados para este modo hoy

    # 2. Ordenar por EV descendente
    candidates.sort(key=lambda c: c.expected_value, reverse=True)

    # 3. Construir el ticket respetando: 1 mercado por partido + sin correlaciones negativas
    selected: list[TicketLegSchema] = []
    selected_match_ids: set[int] = set()
    selected_market_names: list[str] = []

    for candidate in candidates:
        if len(selected) >= max_legs:
            break

        # Regla: máximo 1 mercado por partido
        if candidate.match_id in selected_match_ids:
            continue

        # Verificar que no crea combinación prohibida
        test_markets = selected_market_names + [candidate.market_name]
        is_valid, reason = check_forbidden_combination(test_markets)
        if not is_valid:
            continue

        selected.append(candidate)
        selected_match_ids.add(candidate.match_id)
        selected_market_names.append(candidate.market_name)

    if len(selected) < 2:
        return None

    # 4. Verificar que la cuota combinada esté en el rango objetivo
    combined = calculate_combined_odds(selected)
    if not (target_min <= combined <= target_max):
        # Si está fuera de rango, intentamos ajustar (quitar el leg de mayor cuota si pasado)
        if combined > target_max and len(selected) > 2:
            selected = sorted(selected, key=lambda x: x.bookmaker_odds)[:max_legs - 1]
            combined = calculate_combined_odds(selected)

    # 5. Calcular métricas finales
    avg_ev = calculate_average_ev(selected)
    corr_bonus = get_correlation_bonus(selected_market_names)
    base_confidence = min(
        round(avg_ev * 400 + corr_bonus * 20 + len(selected) * 5), 95
    )

    # 6. Determinar labels por modo
    mode_labels = {
        TicketMode.EDGE:  "EDGE MODE",
        TicketMode.VALUE: "VALUE MODE",
        TicketMode.BOLD:  "BOLD MODE",
    }
    correlation_status = "positive" if corr_bonus > 0.5 else "independent"

    return GeneratedTicket(
        mode=mode,
        mode_label=mode_labels[mode],
        legs=selected,
        combined_odds=combined,
        average_ev=avg_ev,
        confidence_score=base_confidence,
        correlation_validated=True,
        tactical_summary=f"{len(selected)} selections with average {avg_ev*100:.1f}% EV advantage over bookmaker. Correlation: {correlation_status}.",
        pros=[
            f"All legs passed minimum {min_ev*100:.0f}% EV threshold",
            f"No negative correlations detected across {len(selected)} markets",
            f"Combined odds {combined}x within {mode.value} target range",
        ],
        cons=[
            "Past model performance does not guarantee future results",
            f"Lower confidence legs: {sum(1 for l in selected if l.our_probability < 0.55)} selection(s) below 55%",
        ],
        staking_suggestion=config["staking"],
    )
```

**El endpoint:**

```python
# apps/api/routes/v1/tickets.py
from fastapi import APIRouter, Depends, Query
from datetime import date
from zoneinfo import ZoneInfo

from apps.api.schemas.ticket import (
    TicketGenerateRequest,
    TicketGenerateResponse,
    TicketMode,
)
from apps.api.engine.ticket_builder import build_ticket_for_mode
from apps.api.orchestrators.prediction_orchestrator import PredictionOrchestrator
from apps.api.repositories.match_repository import MatchRepository
from apps.api.dependencies import get_async_session, get_cache_service

router = APIRouter(prefix="/tickets", tags=["Tickets"])

COT = ZoneInfo("America/Bogota")


@router.post(
    "/generate",
    response_model=TicketGenerateResponse,
    summary="Generate AI tickets for today's matches (EDGE, VALUE, BOLD)",
)
async def generate_tickets(
    request: TicketGenerateRequest,
    session=Depends(get_async_session),
    cache=Depends(get_cache_service),
):
    """
    Analiza todos los partidos del día, calcula EV para cada mercado
    y genera un ticket por cada modo solicitado.

    Reglas de construcción:
    - Máximo 1 mercado por partido por ticket
    - Ninguna combinación de correlación negativa
    - Cuota combinada dentro del rango objetivo del modo
    - Solo mercados con EV por encima del umbral del modo
    """
    today_cot = date.today()
    cache_key = f"tickets:daily:{today_cot.isoformat()}"

    # Intentar desde caché (TTL 30 minutos — los tickets del día son relativamente estables)
    if cached := await cache.get(cache_key, TicketGenerateResponse):
        # Filtrar por modos solicitados si el usuario quiere solo algunos
        if set(request.modes) != {TicketMode.EDGE, TicketMode.VALUE, TicketMode.BOLD}:
            cached.tickets = [t for t in cached.tickets if t.mode in request.modes]
        return cached

    # Cargar todos los partidos de hoy desde DB
    repo = MatchRepository(session)
    todays_matches = await repo.get_matches_by_date(
        date=today_cot,
        league_keys=request.league_filter,
    )

    if not todays_matches:
        return TicketGenerateResponse(
            generated_at=__import__("datetime").datetime.now(COT).isoformat(),
            tickets=[],
            total_ev_opportunities=0,
            matches_analyzed=0,
        )

    # Obtener predicciones para cada partido (usa caché de predicciones interna)
    orchestrator = PredictionOrchestrator(repo=repo, cache=cache)
    all_predictions = []

    for match in todays_matches:
        try:
            pred = await orchestrator.get_prediction(
                match_id=match.id,
                odds=None,   # Sin cuotas del usuario — usa las de la DB
            )
            all_predictions.append({
                "match_id": match.id,
                "home_team": match.home_team.name,
                "away_team": match.away_team.name,
                "league": match.league.name,
                "match_time_cot": _format_cot_time(match.match_date),
                "markets": [
                    {
                        "market_name":       m.market_name,
                        "market_label":      _market_label(m.market_name),
                        "our_probability":   m.our_probability,
                        "bookmaker_odds":    m.bookmaker_odds or 0,
                        "implied_probability": m.implied_probability or 0,
                        "expected_value":    m.expected_value or -1,
                    }
                    for m in pred.ev_analysis
                    if m.bookmaker_odds and m.expected_value is not None
                ],
            })
        except Exception:
            continue   # Degradación elegante: si un partido falla, el resto sigue

    # Contar oportunidades EV+ totales
    total_ev = sum(
        1 for pred in all_predictions
        for mkt in pred["markets"]
        if mkt["expected_value"] > 0.05
    )

    # Construir un ticket por modo
    tickets = []
    for mode in request.modes:
        ticket = build_ticket_for_mode(mode, all_predictions)
        if ticket:
            tickets.append(ticket)

    response = TicketGenerateResponse(
        generated_at=__import__("datetime").datetime.now(COT).isoformat(),
        tickets=tickets,
        total_ev_opportunities=total_ev,
        matches_analyzed=len(todays_matches),
    )

    # Cachear por 30 minutos
    await cache.set(cache_key, response, ttl=60 * 30)

    return response


def _format_cot_time(dt) -> str:
    """Convierte datetime UTC a string legible en COT."""
    if dt is None:
        return "TBD"
    from datetime import datetime, timezone
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    cot_dt = dt.astimezone(COT)
    return cot_dt.strftime("%-I:%M %p COT")


def _market_label(market_name: str) -> str:
    labels = {
        "1X2_HOME":     "Home Win",
        "1X2_DRAW":     "Draw",
        "1X2_AWAY":     "Away Win",
        "OVER_2_5":     "Over 2.5 Goals",
        "UNDER_2_5":    "Under 2.5 Goals",
        "OVER_1_5":     "Over 1.5 Goals",
        "OVER_3_5":     "Over 3.5 Goals",
        "BTTS_YES":     "BTTS Yes",
        "BTTS_NO":      "BTTS No",
        "CORNERS_OVER": "Corners Over",
        "CARDS_OVER":   "Cards Over",
    }
    return labels.get(market_name, market_name.replace("_", " ").title())
```
